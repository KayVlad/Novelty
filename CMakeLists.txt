cmake_minimum_required(VERSION 3.24)
project(Novelty VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NOVELTY_LOCAL_QT "Use the matching Deck runtime and project-local development headers" OFF)
if(NOVELTY_LOCAL_QT)
  foreach(module Core Gui Widgets Sql Network)
    string(TOUPPER "${module}" module_upper)
    add_library(Qt6::${module} SHARED IMPORTED)
    set_target_properties(Qt6::${module} PROPERTIES
      IMPORTED_LOCATION "/usr/lib/libQt6${module}.so.6"
      INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/.tools/qt-include"
      INTERFACE_COMPILE_DEFINITIONS "QT_${module_upper}_LIB;QT_NO_DEBUG"
      INTERFACE_COMPILE_OPTIONS "-fPIC")
  endforeach()
  set_property(TARGET Qt6::Widgets APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Gui Qt6::Core)
  set_property(TARGET Qt6::Gui APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
  set_property(TARGET Qt6::Sql APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
  set_property(TARGET Qt6::Network APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
else()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Sql Network)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  config/settings.json config/library-settings.json config/recipes/general-game.json config/recipes/general-vn.json)
file(READ "${CMAKE_SOURCE_DIR}/config/settings.json" NOVELTY_SETTINGS_JSON)
file(READ "${CMAKE_SOURCE_DIR}/config/library-settings.json" NOVELTY_LIBRARY_JSON)
file(READ "${CMAKE_SOURCE_DIR}/config/recipes/general-game.json" NOVELTY_GAME_JSON)
file(READ "${CMAKE_SOURCE_DIR}/config/recipes/general-vn.json" NOVELTY_VN_JSON)
configure_file(src/default_config.h.in generated/default_config.h @ONLY)
add_library(novelty_core src/app_settings.cpp src/core.cpp src/recipes.cpp src/runner_manager.cpp src/folder_import.cpp src/installation.cpp src/diagnostics.cpp src/launch.cpp src/setup.cpp src/runtime.cpp src/runtime_migration.cpp src/removal.cpp src/save_internal.cpp src/save_discovery.cpp src/save_snapshots.cpp src/save_restore.cpp src/save_server_store.cpp src/save_bundle_archive.cpp src/save_server_client.cpp src/save_server_config.cpp src/save_server_state.cpp src/save_server_sync.cpp src/steam_shortcut.cpp src/devices.cpp src/pairing.cpp src/peer_tls.cpp src/lan_connection.cpp src/lan_discovery.cpp src/lan_saves.cpp src/lan_sync.cpp src/lan_games.cpp)
target_include_directories(novelty_core PUBLIC src)
target_include_directories(novelty_core PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(novelty_core PUBLIC Qt6::Core Qt6::Sql Qt6::Network OpenSSL::Crypto)
target_compile_options(novelty_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(novelty_ui src/proton_manager.cpp src/window.cpp src/window_common.cpp src/window_settings.cpp src/window_manage.cpp
  src/window_devices.cpp src/window_peers.cpp src/window_launch.cpp src/presentation.cpp src/provisioning.cpp src/metadata.cpp)
target_link_libraries(novelty_ui PUBLIC novelty_core Qt6::Widgets Qt6::Network)
target_compile_options(novelty_ui PRIVATE -Wall -Wextra -Wpedantic)
add_executable(novelty src/main.cpp)
target_link_libraries(novelty PRIVATE novelty_ui)
add_executable(novelty-save-server src/save_server_main.cpp)
target_link_libraries(novelty-save-server PRIVATE novelty_core Qt6::Network OpenSSL::Crypto)
include(CTest)
