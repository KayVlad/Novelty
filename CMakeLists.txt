cmake_minimum_required(VERSION 3.24)
project(Novelty VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NOVELTY_LOCAL_QT "Use the matching Deck runtime and project-local development headers" OFF)
if(NOVELTY_LOCAL_QT)
  foreach(module Core Gui Widgets Sql Network)
    string(TOUPPER "${module}" module_upper)
    add_library(Qt6::${module} SHARED IMPORTED)
    set_target_properties(Qt6::${module} PROPERTIES
      IMPORTED_LOCATION "/usr/lib/libQt6${module}.so.6"
      INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/.tools/qt-include"
      INTERFACE_COMPILE_DEFINITIONS "QT_${module_upper}_LIB;QT_NO_DEBUG"
      INTERFACE_COMPILE_OPTIONS "-fPIC")
  endforeach()
  set_property(TARGET Qt6::Widgets APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Gui Qt6::Core)
  set_property(TARGET Qt6::Gui APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
  set_property(TARGET Qt6::Sql APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
  set_property(TARGET Qt6::Network APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
else()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Sql Network)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(novelty_core src/core.cpp src/folder_import.cpp src/installation.cpp src/diagnostics.cpp src/launch.cpp src/setup.cpp src/profiles.cpp src/runtime.cpp src/runtime_migration.cpp src/removal.cpp src/saves.cpp src/steam_shortcut.cpp src/devices.cpp src/pairing.cpp src/peer_tls.cpp src/lan_connection.cpp src/lan_saves.cpp src/lan_sync.cpp src/lan_games.cpp src/sync_queue.cpp)
target_include_directories(novelty_core PUBLIC src)
target_link_libraries(novelty_core PUBLIC Qt6::Core Qt6::Sql Qt6::Network OpenSSL::Crypto)
target_compile_options(novelty_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(novelty src/main.cpp src/window.cpp src/presentation.cpp src/provisioning.cpp src/metadata.cpp)
target_link_libraries(novelty PRIVATE novelty_core Qt6::Widgets Qt6::Network)
include(CTest)
if(BUILD_TESTING)
  add_executable(novelty_tests tests/core_tests.cpp)
  target_link_libraries(novelty_tests PRIVATE novelty_core Qt6::Network)
  add_test(NAME core COMMAND novelty_tests)
  add_executable(novelty_lan_integration tests/lan_integration.cpp)
  target_link_libraries(novelty_lan_integration PRIVATE novelty_core)
  add_test(NAME lan_integration COMMAND novelty_lan_integration)
  set_tests_properties(lan_integration PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
  add_executable(novelty_real_save_acceptance tests/real_save_acceptance.cpp)
  target_link_libraries(novelty_real_save_acceptance PRIVATE novelty_core)
  add_executable(novelty_ui_tests tests/ui_tests.cpp src/window.cpp src/presentation.cpp src/provisioning.cpp src/metadata.cpp)
  target_link_libraries(novelty_ui_tests PRIVATE novelty_core Qt6::Widgets Qt6::Network)
  add_test(NAME ui_lifecycle COMMAND novelty_ui_tests $<TARGET_FILE:novelty> $<TARGET_FILE:novelty_removal_driver>)
  set_tests_properties(ui_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(novelty_executable_ui_tests tests/executable_ui_tests.cpp src/window.cpp src/presentation.cpp src/provisioning.cpp src/metadata.cpp)
  target_link_libraries(novelty_executable_ui_tests PRIVATE novelty_core Qt6::Widgets Qt6::Network)
  add_test(NAME executable_ui COMMAND novelty_executable_ui_tests $<TARGET_FILE:novelty>)
  set_tests_properties(executable_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME runtime_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/runtime_recovery.py $<TARGET_FILE:novelty>)
  set_tests_properties(runtime_recovery PROPERTIES TIMEOUT 30)
  add_executable(novelty_removal_driver tests/removal_driver.cpp)
  target_link_libraries(novelty_removal_driver PRIVATE novelty_core)
  add_test(NAME removal_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/removal_recovery.py $<TARGET_FILE:novelty> $<TARGET_FILE:novelty_removal_driver>)
  set_tests_properties(removal_recovery PROPERTIES TIMEOUT 30)
  add_test(NAME installers COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_*.py)
  set_tests_properties(installers PROPERTIES TIMEOUT 30)
endif()
install(TARGETS novelty RUNTIME DESTINATION bin)
install(PROGRAMS scripts/install-ge.py scripts/install-umu.py DESTINATION share/novelty)
